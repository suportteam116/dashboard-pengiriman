// ================================================================
// APPS SCRIPT — Dashboard Pengiriman + Investigasi Retur
// Paste SELURUH kode ini ke Google Apps Script (Extensions → Apps Script)
// lalu Deploy → New Deployment → Web App
//   Execute as: Me
//   Who has access: Anyone
// (Kalau sudah pernah deploy, gunakan "Manage deployments" → Edit → New version,
//  supaya URL /exec tetap sama dan tidak perlu ganti SCRIPT_URL di dashboard)
//
// UPDATE (kolom "Nama CS"): ditambahkan kolom "Nama CS" (kolom terakhir,
// index 13) supaya tab "Performa CS" dan tab "Investigasi Retur" di
// dashboard bisa saling terhubung.
//
// UPDATE (anti-duplikat): kalau No. Resi yang disubmit SUDAH ADA di sheet,
// data baru (foto, catatan, bukti chat, dll) akan DIGABUNGKAN ke baris yang
// sudah ada, BUKAN dibuat baris baru. Kalau No. Resi kosong, pencocokan
// pakai Nama Buyer sebagai gantinya. Kalau No. Resi/Nama beda dari data
// manapun yang sudah ada (misal nama sama tapi resi beda), tetap dianggap
// data baru dan dibuat baris baru seperti biasa.
//
// UPDATE (Master Opsi Kurir/Platform/Nama CS/Alasan): dropdown "Kurir",
// "Platform", "Nama CS", dan "Alasan Ditolak/Kendala" sekarang bisa
// ditambah/edit/hapus langsung dari dashboard (tombol "⚙️ Kelola Master …"),
// dan disimpan ke sheet baru "Master Opsi" (bukan cuma localStorage browser),
// jadi pilihannya sinkron di semua perangkat/browser. Sheet ini otomatis
// dibuat saat pertama kali ada perubahan disimpan — tidak perlu dibuat manual.
// ================================================================

// ── NAMA SHEET UNTUK INVESTIGASI RETUR (boleh diganti) ──
var INVESTIGASI_SHEET = 'Investigasi Retur';
var INVESTIGASI_HEADER = [
  'Timestamp', 'Tanggal Kejadian', 'No Resi', 'Kurir', 'Platform',
  'Nama Buyer', 'Alasan Ditolak/Kendala', 'Catatan', 'Bukti Chat (Teks)',
  'Link Bukti Gambar 1', 'Link Bukti Gambar 2', 'Link Bukti Gambar 3', 'Status',
  'Nama CS'
];
var DRIVE_FOLDER_NAME = 'Bukti Investigasi Retur - HBH';

// ── NAMA SHEET UNTUK "KIRIM MANUAL (FOTO RESI)" ──
// Dipakai untuk paket yang ekspedisinya (mis. konter TIKI) cuma kasih bukti
// resi FOTO/fisik, bukan data sistem. Data ini SENGAJA dipisah dari sheet
// Investigasi Retur supaya tidak tercampur (beda tujuan: ini catatan
// pengiriman, bukan catatan retur/kendala).
var MANUAL_SHEET = 'Pengiriman Manual (Foto)';
var MANUAL_HEADER = [
  'Timestamp Input', 'Tanggal Kirim', 'No Resi', 'Kurir', 'Nama Penerima',
  'Alamat Penerima', 'Kota/Kab', 'Kode Pos', 'Telepon Penerima', 'Isi Kiriman',
  'Berat (kg)', 'Ongkir', 'ETA', 'Nama CS', 'Link Foto Resi', 'Catatan'
];
var DRIVE_FOLDER_MANUAL = 'Bukti Foto Resi Manual - HBH';

// ── NAMA SHEET UNTUK "MASTER OPSI" (dropdown Kurir/Platform/Nama CS/Alasan) ──
// Menyimpan pilihan dropdown yang bisa ditambah/edit/hapus langsung dari
// dashboard (tombol "⚙️ Kelola Master …"), TERSIMPAN DI SINI (Google Sheet),
// bukan cuma di localStorage browser — jadi sinkron di semua perangkat, dan
// juga bisa diedit langsung di Google Sheet ini kalau perlu.
var MASTER_SHEET = 'Master Opsi';
var MASTER_HEADER = ['Kategori', 'Nilai'];

// ================================================================
// GET — untuk membaca data sheet apa pun (dipakai dashboard existing
// dan juga dipakai untuk menampilkan riwayat Investigasi Retur)
// ================================================================
function doGet(e) {
  try {
    var params = (e && e.parameter) || {};

    // Nama sheet — sesuaikan jika berbeda
    var sheetName = params.sheet || 'DATA QWARTAL 2';

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);

    // Fallback: coba nama sheet lain yang umum jika tidak ditemukan
    if (!sheet) {
      var allSheets = ss.getSheets().map(function(s){ return s.getName(); });
      for (var i = 0; i < allSheets.length; i++) {
        var n = allSheets[i].toLowerCase();
        if (n.includes('qwartal') || n.includes('data') || n.includes('pengiriman')) {
          sheet = ss.getSheetByName(allSheets[i]);
          break;
        }
      }
    }

    // Kalau yang diminta sheet Investigasi Retur tapi belum pernah dibuat
    // (belum ada submit sama sekali), kembalikan sukses dengan data kosong
    // supaya dashboard tidak error, bukan pesan "sheet tidak ditemukan".
    if (!sheet && sheetName === INVESTIGASI_SHEET) {
      return jsonResponse({ success: true, data: [INVESTIGASI_HEADER], rows: 0 });
    }
    if (!sheet && sheetName === MANUAL_SHEET) {
      return jsonResponse({ success: true, data: [MANUAL_HEADER], rows: 0 });
    }
    if (!sheet && sheetName === MASTER_SHEET) {
      return jsonResponse({ success: true, data: [MASTER_HEADER], rows: 0 });
    }

    if (!sheet) {
      return jsonResponse({
        success: false,
        error: 'Sheet tidak ditemukan. Sheet tersedia: ' +
               SpreadsheetApp.getActiveSpreadsheet().getSheets()
                 .map(function(s){ return s.getName(); }).join(', ')
      });
    }

    var lastRow  = sheet.getLastRow();
    var lastCol  = sheet.getLastColumn();

    if (lastRow < 2) {
      return jsonResponse({ success: false, error: 'Sheet kosong atau tidak ada data.' });
    }

    var data = sheet.getRange(1, 1, lastRow, lastCol).getValues();

    var result = data.map(function(row) {
      return row.map(function(cell) {
        if (cell instanceof Date) {
          if (cell.getTime() === 0 || isNaN(cell.getTime())) return '';
          var d = cell.getDate();
          var mo = cell.getMonth() + 1;
          var y = cell.getFullYear();
          var h = cell.getHours();
          var mi = cell.getMinutes();
          var s = cell.getSeconds();
          return pad(d)+'-'+pad(mo)+'-'+y+' '+pad(h)+':'+pad(mi)+':'+pad(s);
        }
        return cell;
      });
    });

    return jsonResponse({ success: true, data: result, rows: result.length - 1 });

  } catch (err) {
    return jsonResponse({ success: false, error: err.message });
  }
}

// ================================================================
// POST — untuk menerima submit form "Investigasi Retur" dari dashboard
// (catatan teks + sampai 3 gambar bukti dalam bentuk base64 + nama CS)
//
// Dikirim dari dashboard dengan Content-Type: text/plain (BUKAN
// application/json) supaya browser tidak melakukan CORS preflight
// OPTIONS request, yang tidak didukung Apps Script Web App.
//
// LOGIKA ANTI-DUPLIKAT:
//   1. Kalau No. Resi diisi → dicocokkan ke kolom "No Resi" yang sudah ada.
//   2. Kalau No. Resi kosong tapi Nama Buyer diisi → dicocokkan ke baris
//      yang No Resi-nya JUGA kosong dan Nama Buyer-nya sama.
//   3. Kalau ketemu → data baru (foto, catatan, bukti chat, kurir,
//      platform, alasan, CS) DIGABUNGKAN ke baris lama, tidak bikin baris
//      baru. No Resi baris lama tidak pernah diubah (itu kunci pencocokan).
//   4. Kalau tidak ketemu → appendRow seperti biasa (baris baru).
// ================================================================
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ success: false, error: 'Tidak ada data terkirim.' });
    }

    var body = JSON.parse(e.postData.contents);

    // ── Action baru untuk "Kirim Manual (Foto Resi)" — dipisah dari
    // alur addInvestigasi supaya tidak menyentuh sheet Investigasi Retur ──
    if (body.action === 'extractResiFoto') {
      return handleExtractResiFoto(body);
    }
    if (body.action === 'addPengirimanManual') {
      return handleAddPengirimanManual(body);
    }
    // ── Action untuk "Kelola Master" (Kurir/Platform/Nama CS/Alasan) ──
    // Menyimpan SELURUH daftar pilihan satu kategori sekaligus (replace),
    // supaya cocok dengan cara kerja modal "Kelola Master …" di dashboard
    // (edit semua baris lalu klik "Simpan Perubahan" sekali jalan).
    if (body.action === 'saveMasterOptions') {
      return handleSaveMasterOptions(body);
    }

    if (body.action !== 'addInvestigasi') {
      return jsonResponse({ success: false, error: 'Action tidak dikenali.' });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(INVESTIGASI_SHEET);
    if (!sheet) {
      sheet = ss.insertSheet(INVESTIGASI_SHEET);
      sheet.appendRow(INVESTIGASI_HEADER);
      sheet.setFrozenRows(1);
    }

    // Kalau sheet ini sudah pernah dibuat SEBELUM update kolom "Nama CS"
    // (jadi headernya masih 13 kolom), tambahkan header kolom ke-14 secara
    // otomatis supaya data lama tidak perlu diedit manual.
    var lastCol = sheet.getLastColumn();
    if (lastCol < INVESTIGASI_HEADER.length) {
      sheet.getRange(1, INVESTIGASI_HEADER.length).setValue(INVESTIGASI_HEADER[INVESTIGASI_HEADER.length - 1]);
    }

    // Upload gambar bukti ke folder Google Drive khusus
    var imageLinks = [];
    if (body.images && body.images.length) {
      var folder = getOrCreateFolder(DRIVE_FOLDER_NAME);
      body.images.forEach(function(img) {
        if (!img.data) return;
        try {
          var blob = Utilities.newBlob(
            Utilities.base64Decode(img.data),
            img.mimeType || 'image/jpeg',
            (body.noResi || 'bukti') + '_' + (img.filename || 'gambar.jpg')
          );
          var file = folder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          // Simpan link "direct view" (bukan file.getUrl() / share link biasa),
          // supaya kalau link ini diklik langsung dari Google Sheet, yang
          // terbuka gambarnya saja tanpa bingkai UI Google Drive.
          imageLinks.push('https://drive.google.com/uc?export=view&id=' + file.getId());
        } catch (imgErr) {
          // kalau satu gambar gagal, lanjut simpan yang lain, jangan gagalkan semua
          imageLinks.push('ERROR: ' + imgErr.message);
        }
      });
    }

    // ── Cek apakah resi (atau nama, kalau resi kosong) ini sudah ada ──
    var existingRowNum = findExistingInvestigasiRow(sheet, body.noResi, body.nama);

    if (existingRowNum > 0) {
      // Sudah ada → gabungkan ke baris lama, jangan bikin baris baru
      updateExistingInvestigasi(sheet, existingRowNum, body, imageLinks);
      return jsonResponse({ success: true, updated: true, row: existingRowNum, imageLinks: imageLinks });
    }

    // Belum ada → baris baru seperti biasa
    var row = [
      new Date(),                 // 0  Timestamp input
      body.tanggal || '',         // 1  Tanggal Kejadian
      body.noResi || '',          // 2  No Resi
      body.kurir || '',           // 3  Kurir
      body.platform || '',        // 4  Platform
      body.nama || '',            // 5  Nama Buyer
      body.alasan || '',          // 6  Alasan
      body.catatan || '',         // 7  Catatan
      body.buktiChat || '',       // 8  Bukti Chat teks
      imageLinks[0] || '',        // 9  Gambar 1
      imageLinks[1] || '',        // 10 Gambar 2
      imageLinks[2] || '',        // 11 Gambar 3
      'Baru',                     // 12 Status
      body.cs || ''                // 13 Nama CS
    ];
    sheet.appendRow(row);

    return jsonResponse({ success: true, updated: false, imageLinks: imageLinks });

  } catch (err) {
    return jsonResponse({ success: false, error: err.message });
  }
}

// Cari baris existing di sheet Investigasi Retur yang cocok dengan
// resi/nama yang baru disubmit. Return nomor baris (1-based, siap dipakai
// di getRange) kalau ketemu, atau -1 kalau tidak ada yang cocok.
function findExistingInvestigasiRow(sheet, resi, nama) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;

  var values = sheet.getRange(2, 1, lastRow - 1, INVESTIGASI_HEADER.length).getValues();
  var resiNorm = (resi || '').toString().trim().toLowerCase();
  var namaNorm = (nama || '').toString().trim().toLowerCase();

  for (var i = 0; i < values.length; i++) {
    var rowResi = (values[i][2] || '').toString().trim().toLowerCase();
    var rowNama = (values[i][5] || '').toString().trim().toLowerCase();

    if (resiNorm) {
      // No Resi diisi → itu kunci utama pencocokan.
      if (rowResi && rowResi === resiNorm) return i + 2; // +2: lewati header, index 1-based
    } else if (namaNorm) {
      // No Resi kosong → fallback cocokkan lewat Nama Buyer, TAPI hanya
      // dengan baris lama yang No Resi-nya juga kosong (supaya nama yang
      // sama dengan resi berbeda tetap dianggap data/paket berbeda).
      if (!rowResi && rowNama && rowNama === namaNorm) return i + 2;
    }
  }
  return -1;
}

// Gabungkan submit baru ke baris Investigasi Retur yang sudah ada, tanpa
// membuat baris duplikat. Catatan & bukti chat baru ditambahkan (bukan
// menimpa) supaya riwayat follow-up sebelumnya tidak hilang. Foto baru
// mengisi slot Gambar 1/2/3 yang masih kosong; kalau ketiganya sudah
// penuh, foto tambahan ditumpuk di slot Gambar 3 (dipisah baris baru
// dalam sel yang sama) supaya tidak ada bukti yang hilang.
function updateExistingInvestigasi(sheet, rowNum, body, imageLinks) {
  var range = sheet.getRange(rowNum, 1, 1, INVESTIGASI_HEADER.length);
  var existing = range.getValues()[0];
  var stamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd-MM-yyyy HH:mm');

  // Catatan (index 7): tambahkan, jangan timpa
  if (body.catatan) {
    existing[7] = existing[7]
      ? (existing[7] + '\n---\n[' + stamp + '] ' + body.catatan)
      : body.catatan;
  }

  // Bukti Chat teks (index 8): tambahkan, jangan timpa
  if (body.buktiChat) {
    existing[8] = existing[8]
      ? (existing[8] + '\n---\n[' + stamp + '] ' + body.buktiChat)
      : body.buktiChat;
  }

  // Gambar 1/2/3 (index 9,10,11): isi slot kosong dulu, sisanya ditumpuk
  var imgCols = [9, 10, 11];
  var imgPtr = 0;
  imageLinks.forEach(function(link) {
    while (imgPtr < imgCols.length && existing[imgCols[imgPtr]]) imgPtr++;
    if (imgPtr < imgCols.length) {
      existing[imgCols[imgPtr]] = link;
      imgPtr++;
    } else {
      var lastImgCol = imgCols[imgCols.length - 1];
      existing[lastImgCol] = existing[lastImgCol] + '\n' + link;
    }
  });

  // Field lain: isi kalau sebelumnya kosong ATAU update kalau user
  // memang mengisi ulang dengan nilai baru. No Resi (index 2) SENGAJA
  // tidak pernah diubah karena itu kunci pencocokan.
  if (body.tanggal)  existing[1] = body.tanggal;
  if (body.kurir)    existing[3] = body.kurir;
  if (body.platform) existing[4] = body.platform;
  if (body.nama)     existing[5] = body.nama;
  if (body.alasan)   existing[6] = body.alasan;
  if (body.cs)       existing[13] = body.cs;
  if (!existing[12]) existing[12] = 'Baru'; // Status: jangan timpa kalau sudah pernah diisi manual

  range.setValues([existing]);
}

// ================================================================
// "KIRIM MANUAL (FOTO RESI)" — dua action baru, TERPISAH dari
// Investigasi Retur:
//
//  1. extractResiFoto  → terima 1 foto (base64), kirim ke Gemini API
//     (model vision) lewat UrlFetchApp (server-side, API key aman di
//     Script Properties), balikin hasil bacaan resi sebagai JSON.
//     TIDAK menulis apa pun ke sheet — cuma untuk auto-isi form.
//  2. addPengirimanManual → simpan data yang SUDAH DICEK/DIKONFIRMASI
//     user ke sheet "Pengiriman Manual (Foto)" (beda sheet dari
//     Investigasi Retur), termasuk upload foto resi ke folder Drive
//     terpisah.
//
// SETUP WAJIB SEKALI: buka Project Settings (⚙️) di editor Apps Script
// → Script Properties → tambahkan properti GEMINI_API_KEY dengan API
// key dari aistudio.google.com (Google AI Studio → Get API key, GRATIS).
// Tanpa ini, tombol "Baca Otomatis dari Foto" akan gagal.
// ================================================================
function handleExtractResiFoto(body) {
  try {
    if (!body.image || !body.image.data) {
      return jsonResponse({ success: false, error: 'Foto tidak ada.' });
    }
    var apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
    if (!apiKey) {
      return jsonResponse({ success: false, error: 'GEMINI_API_KEY belum diatur di Script Properties (Project Settings).' });
    }

    var prompt = 'Ini foto resi/label pengiriman ekspedisi (JNE/TIKI/POS/J&T/SPX/ID Express/dll). ' +
      'Baca semua teks yang tercetak di resi ini, lalu balas HANYA dengan JSON (tanpa markdown, tanpa penjelasan apa pun) ' +
      'dengan struktur PERSIS seperti ini: ' +
      '{"noResi":"","kurir":"","namaPenerima":"","alamat":"","kotaKab":"","kodePos":"","telepon":"","isiKiriman":"","berat":"","ongkir":"","tanggalKirim":"","eta":""}. ' +
      'Aturan: tanggalKirim & eta format YYYY-MM-DD (kalau di resi formatnya DD-MM-YYYY, konversi ke YYYY-MM-DD). ' +
      'kurir cukup nama ekspedisinya saja (contoh: TIKI, JNE, POS, SPX). berat & ongkir hanya angka tanpa satuan/simbol Rp/koma. ' +
      'Kalau ada field yang tidak terbaca/tidak ada di foto, isi string kosong "". JANGAN mengarang data yang tidak ada di foto.';

    var payload = {
      contents: [{
        role: 'user',
        parts: [
          { inline_data: { mime_type: body.image.mimeType || 'image/jpeg', data: body.image.data } },
          { text: prompt }
        ]
      }],
      generationConfig: { responseMimeType: 'application/json' }
    };

    var GEMINI_MODEL = 'gemini-3.6-flash';
    var resp = UrlFetchApp.fetch(
      'https://generativelanguage.googleapis.com/v1beta/models/' + GEMINI_MODEL + ':generateContent?key=' + apiKey,
      {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify(payload),
        muteHttpExceptions: true
      }
    );

    var code = resp.getResponseCode();
    var text = resp.getContentText();
    if (code < 200 || code >= 300) {
      return jsonResponse({ success: false, error: 'Gemini API error (' + code + '): ' + text.substring(0, 300) });
    }

    var json = JSON.parse(text);
    var cand = json.candidates && json.candidates[0];
    var raw = (cand && cand.content && cand.content.parts && cand.content.parts[0] && cand.content.parts[0].text) || '';
    raw = raw.replace(/```json|```/g, '').trim();

    var extracted;
    try {
      extracted = JSON.parse(raw);
    } catch (parseErr) {
      return jsonResponse({ success: false, error: 'Gagal membaca hasil ekstraksi: ' + raw.substring(0, 200) });
    }

    return jsonResponse({ success: true, extracted: extracted });
  } catch (err) {
    return jsonResponse({ success: false, error: err.message });
  }
}

function handleAddPengirimanManual(body) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(MANUAL_SHEET);
    if (!sheet) {
      sheet = ss.insertSheet(MANUAL_SHEET);
      sheet.appendRow(MANUAL_HEADER);
      sheet.setFrozenRows(1);
    }

    var fotoLink = '';
    if (body.image && body.image.data) {
      try {
        var folder = getOrCreateFolder(DRIVE_FOLDER_MANUAL);
        var blob = Utilities.newBlob(
          Utilities.base64Decode(body.image.data),
          body.image.mimeType || 'image/jpeg',
          (body.noResi || 'resi') + '_foto.jpg'
        );
        var file = folder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        fotoLink = 'https://drive.google.com/uc?export=view&id=' + file.getId();
      } catch (imgErr) {
        fotoLink = 'ERROR: ' + imgErr.message;
      }
    }

    var row = [
      new Date(),                    // 0  Timestamp Input
      body.tanggalKirim || '',       // 1  Tanggal Kirim
      body.noResi || '',             // 2  No Resi
      body.kurir || '',              // 3  Kurir
      body.namaPenerima || '',       // 4  Nama Penerima
      body.alamat || '',             // 5  Alamat Penerima
      body.kotaKab || '',            // 6  Kota/Kab
      body.kodePos || '',            // 7  Kode Pos
      body.telepon || '',            // 8  Telepon Penerima
      body.isiKiriman || '',         // 9  Isi Kiriman
      body.berat || '',              // 10 Berat (kg)
      body.ongkir || '',             // 11 Ongkir
      body.eta || '',                // 12 ETA
      body.cs || '',                 // 13 Nama CS
      fotoLink,                      // 14 Link Foto Resi
      body.catatan || ''             // 15 Catatan
    ];
    sheet.appendRow(row);

    return jsonResponse({ success: true, fotoLink: fotoLink });
  } catch (err) {
    return jsonResponse({ success: false, error: err.message });
  }
}

// Menyimpan/mengganti seluruh daftar pilihan satu kategori (kurir/platform/
// cs/alasan) di sheet "Master Opsi". Baris lama untuk kategori itu dihapus
// dulu, lalu baris baru ditulis sesuai urutan yang dikirim dari dashboard.
function handleSaveMasterOptions(body) {
  try {
    var kategori = (body.kategori || '').toString().trim();
    if (!kategori) return jsonResponse({ success: false, error: 'Kategori tidak ada.' });

    var values = Array.isArray(body.values) ? body.values : [];

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(MASTER_SHEET);
    if (!sheet) {
      sheet = ss.insertSheet(MASTER_SHEET);
      sheet.appendRow(MASTER_HEADER);
      sheet.setFrozenRows(1);
    }

    var lastRow = sheet.getLastRow();
    if (lastRow >= 2) {
      var data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
      // hapus dari bawah ke atas supaya nomor baris yang belum diproses tidak bergeser
      for (var i = data.length - 1; i >= 0; i--) {
        if (String(data[i][0]).toLowerCase() === kategori.toLowerCase()) {
          sheet.deleteRow(i + 2);
        }
      }
    }

    var cleaned = values
      .map(function (v) { return String(v).trim(); })
      .filter(function (v) { return v; });

    cleaned.forEach(function (v) {
      sheet.appendRow([kategori, v]);
    });

    return jsonResponse({ success: true, kategori: kategori, values: cleaned });
  } catch (err) {
    return jsonResponse({ success: false, error: err.message });
  }
}

function getOrCreateFolder(name) {
  var folders = DriveApp.getFoldersByName(name);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(name);
}
